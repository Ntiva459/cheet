Setup Every snippet assumes: 
#include <stdio.h>
#include <stdlib.h>
// Lists & queues: wrapper struct with head/tail, pass List * / Queue *.

1. Singly Linked List

List struct head + tail live in a wrapper struct, pass List *l. Insert end is O(1) thanks to tail; delete end is still O(n) (need the previous node). Rule: every insert/delete that touches the first or last node must fix head/tail.
typedef struct Node {
    int data;
    struct Node *next;
} Node;
typedef struct {
    Node *head, *tail;
} List;
void init(List *l) { l->head = l->tail = NULL; }   // empty list

Insertion

void insertFirst(List *l, int v) {
    Node *n = malloc(sizeof(Node));
    n->data = v;
    n->next = l->head;
    l->head = n;
    if (!l->tail) l->tail = n;      // list was empty -> also the tail
}
void insertEnd(List *l, int v) {
    Node *n = malloc(sizeof(Node));
    n->data = v; n->next = NULL;
    if (!l->tail) l->head = l->tail = n;   // empty list
    else { l->tail->next = n; l->tail = n; }
}
// insert AFTER node p (find p with search first)
void insertAfter(List *l, Node *p, int v) {
    if (!p) return;
    Node *n = malloc(sizeof(Node));
    n->data = v;
    n->next = p->next;              // 1) new -> rest (order matters!)
    p->next = n;                    // 2) p -> new
    if (p == l->tail) l->tail = n;  // p was last -> new tail
}
// insert v AFTER the node whose data == x  (by value)
Node *search(List *l, int v);       // prototype: search is defined below
void insertAfterValue(List *l, int x, int v) {
    insertAfter(l, search(l, x), v);   // search returns NULL if x not found
}

Deletion

void deleteFirst(List *l) {
    if (!l->head) return;
    Node *t = l->head;
    l->head = t->next;
    if (!l->head) l->tail = NULL;   // list became empty
    free(t);
}
void deleteEnd(List *l) {
    if (!l->head) return;
    if (l->head == l->tail) {       // single node
        free(l->head);
        l->head = l->tail = NULL;
        return;
    }
    Node *p = l->head;
    while (p->next != l->tail) p = p->next;   // node BEFORE tail
    free(l->tail);
    l->tail = p;
    p->next = NULL;
}
// delete the node AFTER p
void deleteAfter(List *l, Node *p) {
    if (!p || !p->next) return;
    Node *t = p->next;
    p->next = t->next;
    if (t == l->tail) l->tail = p;  // deleted the last one
    free(t);
}
// delete the first node whose data == v   (BY VALUE)
void deleteValue(List *l, int v) {
    Node *p = l->head, *prev = NULL;
    while (p && p->data != v) { prev = p; p = p->next; }
    if (!p) return;                       // not found
    if (prev) prev->next = p->next;       // middle / last
    else l->head = p->next;               // it was the first
    if (p == l->tail) l->tail = prev;     // removed the last (prev is NULL if it was the only node)
    free(p);
}
// delete the node AFTER the node whose data == x
void deleteAfterValue(List *l, int x) {
    deleteAfter(l, search(l, x));         // deleteAfter ignores NULL
}

Search, display, free

Node *search(List *l, int v) {
    for (Node *p = l->head; p; p = p->next)
        if (p->data == v) return p;
    return NULL;
}
void display(List *l) {
    for (Node *p = l->head; p; p = p->next) printf("%d -> ", p->data);
    printf("NULL\n");
}
void freeList(List *l) {
    Node *p = l->head;
    while (p) { Node *t = p; p = p->next; free(t); }
    l->head = l->tail = NULL;       // don't leave dangling pointers
}

2. Doubly Linked List

Bonus with head + tail, delete end is O(1) and you can display backwards from the tail. Cost: more pointers to fix (prev and next, plus head/tail).
typedef struct DNode {
    int data;
    struct DNode *prev, *next;
} DNode;
typedef struct {
    DNode *head, *tail;
} DList;
void init(DList *l) { l->head = l->tail = NULL; }

Insertion

void insertFirst(DList *l, int v) {
    DNode *n = malloc(sizeof(DNode));
    n->data = v; n->prev = NULL; n->next = l->head;
    if (l->head) l->head->prev = n; else l->tail = n;
    l->head = n;
}
void insertEnd(DList *l, int v) {
    DNode *n = malloc(sizeof(DNode));
    n->data = v; n->next = NULL; n->prev = l->tail;
    if (l->tail) l->tail->next = n; else l->head = n;
    l->tail = n;
}
void insertAfter(DList *l, DNode *p, int v) {
    if (!p) return;
    DNode *n = malloc(sizeof(DNode));
    n->data = v; n->prev = p; n->next = p->next;
    if (p->next) p->next->prev = n; else l->tail = n;
    p->next = n;
}
// insert v AFTER the node whose data == x  (by value)
DNode *search(DList *l, int v);     // prototype: search is defined below
void insertAfterValue(DList *l, int x, int v) {
    insertAfter(l, search(l, x), v);
}

Deletion

void deleteFirst(DList *l) {
    if (!l->head) return;
    DNode *t = l->head;
    l->head = t->next;
    if (l->head) l->head->prev = NULL; else l->tail = NULL;
    free(t);
}
void deleteEnd(DList *l) {          // O(1)!
    if (!l->tail) return;
    DNode *t = l->tail;
    l->tail = t->prev;
    if (l->tail) l->tail->next = NULL; else l->head = NULL;
    free(t);
}
void deleteAfter(DList *l, DNode *p) {
    if (!p || !p->next) return;
    DNode *t = p->next;
    p->next = t->next;
    if (t->next) t->next->prev = p; else l->tail = p;
    free(t);
}
// delete the first node whose data == v   (BY VALUE)
void deleteValue(DList *l, int v) {
    DNode *p = search(l, v);
    if (!p) return;                                        // not found
    if (p->prev) p->prev->next = p->next; else l->head = p->next;   // fix left side
    if (p->next) p->next->prev = p->prev; else l->tail = p->prev;   // fix right side
    free(p);
}
// delete the node AFTER the node whose data == x
void deleteAfterValue(DList *l, int x) {
    deleteAfter(l, search(l, x));
}

Search, display, free

DNode *search(DList *l, int v) {
    for (DNode *p = l->head; p; p = p->next)
        if (p->data == v) return p;
    return NULL;
}
void displayFwd(DList *l)  { for (DNode *p = l->head; p; p = p->next) printf("%d <-> ", p->data); printf("NULL\n"); }
void displayBack(DList *l) { for (DNode *p = l->tail; p; p = p->prev) printf("%d <-> ", p->data); printf("NULL\n"); }
void freeList(DList *l) {
    DNode *p = l->head;
    while (p) { DNode *t = p; p = p->next; free(t); }
    l->head = l->tail = NULL;
}

3. Stack (LIFO)

All O(1) push · pop · peek · isEmpty · isFull

Array version

#define MAX 100
int st[MAX], top = -1;               // top = -1 means empty
int isEmpty(void) { return top == -1; }
int isFull(void)  { return top == MAX - 1; }
void push(int v) {
    if (isFull()) { printf("Overflow\n"); return; }
    st[++top] = v;                   // increment THEN store
}
int pop(void) {
    if (isEmpty()) { printf("Underflow\n"); return -1; }
    return st[top--];                // return THEN decrement
}
int peek(void) { return isEmpty() ? -1 : st[top]; }
void display(void) { for (int i = top; i >= 0; i--) printf("%d\n", st[i]); }

Linked-list version (no size limit)

typedef struct Node { int data; struct Node *next; } Node;
// Node *top = NULL;  in main -> pass &top (double pointer, top changes!)
void push(Node **top, int v) {       // = insertFirst
    Node *n = malloc(sizeof(Node));
    n->data = v; n->next = *top;
    *top = n;
}
int pop(Node **top) {                // = deleteFirst
    if (!*top) { printf("Underflow\n"); return -1; }
    Node *t = *top; int v = t->data;
    *top = t->next;
    free(t);
    return v;
}
int peek(Node *top) { return top ? top->data : -1; }   // read only -> single pointer is enough
int isEmpty(Node *top) { return top == NULL; }
void freeStack(Node **top) { while (*top) pop(top); }

4. Queue (FIFO)

All O(1) enqueue at rear, dequeue from front. Use a circular array or you waste space.

Circular array version

#define MAX 5
int q[MAX], front = 0, rear = -1, count = 0;
int isEmpty(void) { return count == 0; }
int isFull(void)  { return count == MAX; }
void enqueue(int v) {
    if (isFull()) { printf("Full\n"); return; }
    rear = (rear + 1) % MAX;         // wrap around
    q[rear] = v; count++;
}
int dequeue(void) {
    if (isEmpty()) { printf("Empty\n"); return -1; }
    int v = q[front];
    front = (front + 1) % MAX; count--;
    return v;
}
int peek(void) { return isEmpty() ? -1 : q[front]; }
void display(void) { for (int i = 0, j = front; i < count; i++, j = (j + 1) % MAX) printf("%d ", q[j]); }

Linked-list version (Queue struct with head + tail)

typedef struct Node { int data; struct Node *next; } Node;
typedef struct { Node *head, *tail; } Queue;    // head = front, tail = rear
void init(Queue *q) { q->head = q->tail = NULL; }
int isEmpty(Queue *q) { return q->head == NULL; }
void enqueue(Queue *q, int v) {      // insert at tail, O(1)
    Node *n = malloc(sizeof(Node));
    n->data = v; n->next = NULL;
    if (q->tail) q->tail->next = n; else q->head = n;   // empty -> head too
    q->tail = n;
}
int dequeue(Queue *q) {              // remove from head, O(1)
    if (!q->head) { printf("Empty\n"); return -1; }
    Node *t = q->head; int v = t->data;
    q->head = t->next;
    if (!q->head) q->tail = NULL;    // became empty: reset tail!
    free(t);
    return v;
}
int peek(Queue *q) { return q->head ? q->head->data : -1; }
void display(Queue *q) { for (Node *p = q->head; p; p = p->next) printf("%d ", p->data); printf("\n"); }
void freeQueue(Queue *q) { while (q->head) dequeue(q); }

5. Hash Tables

Avg O(1) worst O(n). Hash: h(k) = k % SIZE. Choose SIZE prime to spread keys. Load factor α = n / SIZE.

A) Separate chaining (array of linked lists)

#define SIZE 10
typedef struct Node { int key; struct Node *next; } Node;
Node *table[SIZE];                   // global = all NULL
int hash(int k) { return k % SIZE; }

void insert(int k) {
    int i = hash(k);
    Node *n = malloc(sizeof(Node));
    n->key = k; n->next = table[i];  // insert at head of bucket
    table[i] = n;
}
Node *search(int k) {
    for (Node *p = table[hash(k)]; p; p = p->next)
        if (p->key == k) return p;
    return NULL;
}
void delete(int k) {
    int i = hash(k);
    Node *p = table[i], *prev = NULL;
    while (p && p->key != k) { prev = p; p = p->next; }
    if (!p) return;                  // not found
    if (prev) prev->next = p->next; else table[i] = p->next;
    free(p);
}
void display(void) {
    for (int i = 0; i < SIZE; i++) {
        printf("[%d]: ", i);
        for (Node *p = table[i]; p; p = p->next) printf("%d -> ", p->key);
        printf("NULL\n");
    }
}
void freeTable(void) {
    for (int i = 0; i < SIZE; i++) {
        Node *p = table[i];
        while (p) { Node *t = p; p = p->next; free(t); }
        table[i] = NULL;
    }
}

B) Open addressing (linear probing)

Delete must write a DELETED marker (tombstone), NOT EMPTY — otherwise search stops early and loses keys behind it. Quadratic probing: use (h + i*i) % SIZE. Double hashing: (h1 + i*h2) % SIZE.
#define SIZE 10
#define EMPTY   -1
#define DELETED -2
int t[SIZE];
void init(void) { for (int i = 0; i < SIZE; i++) t[i] = EMPTY; }  // call first!
int hash(int k) { return k % SIZE; }

int insert(int k) {
    int h = hash(k);
    for (int i = 0; i < SIZE; i++) {
        int idx = (h + i) % SIZE;        // linear probe
        if (t[idx] == EMPTY || t[idx] == DELETED) { t[idx] = k; return idx; }
    }
    return -1;                           // table full
}
int search(int k) {
    int h = hash(k);
    for (int i = 0; i < SIZE; i++) {
        int idx = (h + i) % SIZE;
        if (t[idx] == EMPTY) return -1;  // hole = key can't be further
        if (t[idx] == k) return idx;     // DELETED is skipped
    }
    return -1;
}
void delete(int k) {
    int idx = search(k);
    if (idx != -1) t[idx] = DELETED;
}
void display(void) {
    for (int i = 0; i < SIZE; i++) {
        if (t[i] == EMPTY) printf("[%d]: -\n", i);
        else if (t[i] == DELETED) printf("[%d]: deleted\n", i);
        else printf("[%d]: %d\n", i, t[i]);
    }
}
// free: nothing to free (static array). If malloc'd: free(t);

6. Sorting Algorithms

AlgorithmBestAverageWorstSpaceStable
BubbleO(n)*O(n²)O(n²)O(1)yes
SelectionO(n²)O(n²)O(n²)O(1)no
InsertionO(n)O(n²)O(n²)O(1)yes
MergeO(n log n)O(n log n)O(n log n)O(n)yes
*with the swapped early-exit flag.

Bubble sort — swap neighbours, biggest "bubbles" to the end

void bubbleSort(int a[], int n) {
    for (int i = 0; i < n - 1; i++) {
        int swapped = 0;
        for (int j = 0; j < n - 1 - i; j++)      // last i are already sorted
            if (a[j] > a[j + 1]) {
                int t = a[j]; a[j] = a[j + 1]; a[j + 1] = t;
                swapped = 1;
            }
        if (!swapped) break;                    // already sorted -> stop
    }
}

Selection sort — find the minimum, put it in front

void selectionSort(int a[], int n) {
    for (int i = 0; i < n - 1; i++) {
        int min = i;
        for (int j = i + 1; j < n; j++)
            if (a[j] < a[min]) min = j;
        if (min != i) { int t = a[i]; a[i] = a[min]; a[min] = t; }
    }
}

Insertion sort — like sorting cards in your hand

void insertionSort(int a[], int n) {
    for (int i = 1; i < n; i++) {
        int key = a[i], j = i - 1;
        while (j >= 0 && a[j] > key) {      // shift bigger elements right
            a[j + 1] = a[j];
            j--;
        }
        a[j + 1] = key;                        // drop key in the gap
    }
}

Merge sort — divide, sort halves, merge

void merge(int a[], int l, int m, int r) {
    int n1 = m - l + 1, n2 = r - m;
    int L[n1], R[n2];                          // temp copies of both halves
    for (int i = 0; i < n1; i++) L[i] = a[l + i];
    for (int j = 0; j < n2; j++) R[j] = a[m + 1 + j];
    int i = 0, j = 0, k = l;
    while (i < n1 && j < n2)
        a[k++] = (L[i] <= R[j]) ? L[i++] : R[j++];   // <= keeps it stable
    while (i < n1) a[k++] = L[i++];
    while (j < n2) a[k++] = R[j++];
}
void mergeSort(int a[], int l, int r) {        // call: mergeSort(a, 0, n-1)
    if (l < r) {
        int m = l + (r - l) / 2;
        mergeSort(a, l, m);
        mergeSort(a, m + 1, r);
        merge(a, l, m, r);
    }
}
