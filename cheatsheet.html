Singly LL

Every insert/delete at an end must update head/tail. deleteend is O(n): you need the node BEFORE tail. Never copy head/tail into locals and change the copies. Free what you remove.
Given types
typedef struct Node{int data;struct Node*next;}Node;
typedef struct{Node*head,*tail;}List;
insertbegin

Insert val as the NEW FIRST node. Careful: if the list was empty, tail must be set too.

void insertbegin(List *list,int val) {
    Node*n=malloc(sizeof(Node));
    n->data=val;
    n->next=list->head;
    list->head=n;
    if(!list->tail)list->tail=n;
}
insertEnd

Insert val as the LAST node in O(1) using tail. Empty list: head AND tail.

void insertEnd(List *list,int val) {
    Node*n=malloc(sizeof(Node));
    n->data=val;
    n->next=NULL;
    if(list->tail)list->tail->next=n;
    else list->head=n;
    list->tail=n;
}
insertmidd (after value)

Insert val AFTER the first node holding data. If data is not found, append at the end. Update tail if needed.

void insertmidd(List *list,int data,int val) {
    Node*n=malloc(sizeof(Node));
    n->data=val;
    Node*p=list->head;
    while(p&&p->data!=data)p=p->next;
    if(!p){
        n->next=NULL;
        if(list->tail)list->tail->next=n;
        else list->head=n;
        list->tail=n;
        return;
    }
    n->next=p->next;
    p->next=n;
    if(p==list->tail)list->tail=n;
}
deletebegin

Delete the FIRST node (free it!). If the list becomes empty, tail must become NULL.

void deletebegin(List *list) {
    if(!list->head)return;
    Node*t=list->head;
    list->head=t->next;
    if(!list->head)list->tail=NULL;
    free(t);
}
deleteend

Delete the LAST node. Needs the node BEFORE tail. Work on list->head / list->tail, not local copies! Single node case too.

void deleteend(List *list) {
    Node*h=list->head;
    if(!h)return;
    if(h==list->tail){
        free(h);
        list->head=list->tail=NULL;
        return;
    }
    while(h->next!=list->tail)h=h->next;
    free(list->tail);
    list->tail=h;
    h->next=NULL;
}
deletespecific (by value)

Delete the first node whose data == val (do nothing if absent). Cases: head, middle, tail.

void deletespecific(List *list,int val) {
    Node*c=list->head,*prev=NULL;
    while(c&&c->data!=val){
        prev=c;
        c=c->next;
    }
    if(!c)return;
    if(prev)prev->next=c->next;
    else list->head=c->next;
    if(c==list->tail)list->tail=prev;
    free(c);
}
search

Return the first node with data == val, or NULL if absent.

Node *search(List *list,int val) {
    Node*c=list->head;
    while(c&&c->data!=val)c=c->next;
    return c;
}
printList

Print every node as "v -> " and finish with NULL and a newline. Format exactly: 3 -> 5 -> NULL

void printList(List *list) {
    for(Node*c=list->head;c;c=c->next)printf("%d -> ",c->data);
    printf("NULL");
    puts("");
}
freelist

Free EVERY node (save next before freeing!), then set head and tail to NULL.

void freelist(List *list) {
    Node*c=list->head;
    while(c){
        Node*t=c;
        c=c->next;
        free(t);
    }
    list->head=list->tail=NULL;
}
deleteafter (after value)

Find the node holding data and delete the node AFTER it. Nothing to delete if data is absent or it is the last node. If the deleted node was the tail, tail = that node.

void deleteafter(List *list,int data) {
    Node*p=list->head;
    while(p&&p->data!=data)p=p->next;
    if(!p||!p->next)return;
    Node*t=p->next;
    p->next=t->next;
    if(t==list->tail)list->tail=p;
    free(t);
}

Doubly LL

Fix prev AND next (4 links on insert-after). Check p->next / p->prev exist before touching them, otherwise update tail/head. With a tail pointer, delete-last is O(1).
Given types
typedef struct Node{int data;struct Node*next,*prev;}Node;
typedef struct{Node*head,*tail;}List;
insertlast

Insert at the END of the doubly linked list: set prev, next, and head/tail.

void insertlast(List *list,int val) {
    Node*n=malloc(sizeof(Node));
    n->data=val;
    n->next=NULL;
    n->prev=list->tail;
    if(list->tail)list->tail->next=n;
    else list->head=n;
    list->tail=n;
}
insertsomewhere (after node p)

Insert val AFTER node p. 4 links to fix; if p was the tail, update tail.

void insertsomewhere(List *list,Node *p,int val) {
    Node*n=malloc(sizeof(Node));
    n->data=val;
    n->prev=p;
    n->next=p->next;
    if(p->next)p->next->prev=n;
    else list->tail=n;
    p->next=n;
}
deleteValue

Delete first node with data == val. Fix prev->next AND next->prev (head/tail at the edges), then free.

void deleteValue(List *list,int val) {
    Node*p=list->head;
    while(p&&p->data!=val)p=p->next;
    if(!p)return;
    if(p->prev)p->prev->next=p->next;
    else list->head=p->next;
    if(p->next)p->next->prev=p->prev;
    else list->tail=p->prev;
    free(p);
}
insertfirst

Insert at the START. Old head->prev must point to the new node; if the list was empty, tail too.

void insertfirst(List *list,int val) {
    Node*n=malloc(sizeof(Node));
    n->data=val;
    n->prev=NULL;
    n->next=list->head;
    if(list->head)list->head->prev=n;
    else list->tail=n;
    list->head=n;
}
deletefirst

Delete the first node. New head->prev = NULL; if the list becomes empty, tail = NULL.

void deletefirst(List *list) {
    if(!list->head)return;
    Node*t=list->head;
    list->head=t->next;
    if(list->head)list->head->prev=NULL;
    else list->tail=NULL;
    free(t);
}
deletelast

O(1) thanks to tail and prev: tail = tail->prev. Then tail->next = NULL, or head = NULL if it was the only node.

void deletelast(List *list) {
    if(!list->tail)return;
    Node*t=list->tail;
    list->tail=t->prev;
    if(list->tail)list->tail->next=NULL;
    else list->head=NULL;
    free(t);
}
displayforward + displaybackward

Forward: head via next. Backward: tail via prev. Format "v <-> " per node, then NULL.

void displayforward(List *list) {
    for(Node*c=list->head;c;c=c->next)printf("%d <-> ",c->data);
    printf("NULL");
    puts("");
}

void displaybackward(List *list) {
    for(Node*c=list->tail;c;c=c->prev)printf("%d <-> ",c->data);
    printf("NULL");
    puts("");
}
search

Return the first node with data == val, or NULL.

Node *search(List *list,int val) {
    Node*c=list->head;
    while(c&&c->data!=val)c=c->next;
    return c;
}
freelist

Free every node (save next first), then head = tail = NULL.

void freelist(List *list) {
    Node*c=list->head;
    while(c){
        Node*t=c;
        c=c->next;
        free(t);
    }
    list->head=list->tail=NULL;
}
insertmidd (after value)

Insert val AFTER the first node holding data (4 links). If data is not found, append at the end.

void insertmidd(List *list,int data,int val) {
    Node*n=malloc(sizeof(Node));
    n->data=val;
    Node*p=list->head;
    while(p&&p->data!=data)p=p->next;
    if(!p){
        n->next=NULL;
        n->prev=list->tail;
        if(list->tail)list->tail->next=n;
        else list->head=n;
        list->tail=n;
        return;
    }
    n->prev=p;
    n->next=p->next;
    if(p->next)p->next->prev=n;
    else list->tail=n;
    p->next=n;
}
deleteafter (after value)

Find the node holding data and delete the node AFTER it: p->next = t->next, and t->next->prev = p (or tail = p if t was last).

void deleteafter(List *list,int data) {
    Node*p=list->head;
    while(p&&p->data!=data)p=p->next;
    if(!p||!p->next)return;
    Node*t=p->next;
    p->next=t->next;
    if(t->next)t->next->prev=p;
    else list->tail=p;
    free(t);
}

Stack

LIFO. top changes, so push/pop take StackNode **top. peek/isempty only read: single pointer. Never dereference NULL on an empty stack.
Given types
typedef struct StackNode{int val;struct StackNode*next;}StackNode;
push

Push val on top. Double pointer, because top itself changes.

void push(StackNode **top,int val) {
    StackNode*n=malloc(sizeof(StackNode));
    n->val=val;
    n->next=*top;
    *top=n;
}
pop

Remove the top node, free it, return its value.

int pop(StackNode **top) {
    StackNode*t=*top;
    int v=t->val;
    *top=t->next;
    free(t);
    return v;
}
isempty + peek

isempty: top == NULL. peek: value on top WITHOUT removing it (single pointer: nothing changes).

int isempty(StackNode *top) {
    return top==NULL;
}

int peek(StackNode *top) {
    return top->val;
}
displaystack + freestack

display: print values from top down, space separated, then newline. free: free every node and set *top = NULL.

void displaystack(StackNode *top) {
    for(StackNode*c=top;c;c=c->next)printf("%d ",c->val);
    puts("");
}

void freestack(StackNode **top) {
    while(*top){
        StackNode*t=*top;
        *top=t->next;
        free(t);
    }
}

Queue

FIFO: enqueue at rear, dequeue from front. Empty queue: front = rear = new node. After the LAST dequeue set rear = NULL. No malloc inside dequeue.
Given types
typedef struct QueNode{int val;struct QueNode*next;}QueNode;
typedef struct{QueNode*front,*rear;}Queue;
enqueue

Add at the REAR. Empty queue: front = rear = new node.

void enqueue(Queue *q,int val) {
    QueNode*n=malloc(sizeof(QueNode));
    n->val=val;
    n->next=NULL;
    if(q->rear)q->rear->next=n;
    else q->front=n;
    q->rear=n;
}
dequeue

Remove from the FRONT, return the value. If the queue becomes empty, rear must become NULL. Do not malloc here!

int dequeue(Queue *q) {
    QueNode*t=q->front;
    int v=t->val;
    q->front=t->next;
    if(!q->front)q->rear=NULL;
    free(t);
    return v;
}
isemptyque + peekque

isemptyque: front == NULL. peekque: value at the front, without removing.

int isemptyque(Queue *p) {
    return p->front==NULL;
}

int peekque(Queue *p) {
    return p->front->val;
}
displayque + freequeue

display: values from front to rear, space separated, then newline. free: free every node and set front = rear = NULL.

void displayque(Queue *p) {
    for(QueNode*c=p->front;c;c=c->next)printf("%d ",c->val);
    puts("");
}

void freequeue(Queue *p) {
    QueNode*c=p->front;
    while(c){
        QueNode*t=c;
        c=c->next;
        free(t);
    }
    p->front=p->rear=NULL;
}

Array stack

LIFO in an array. top = -1 means empty; full when top == MAX-1. push: ++top then store. pop: return then top--. Always check overflow/underflow first.
Given types
#define MAX 5
int st[MAX];
int top=-1;   // empty
push + pop + peek (array)

top starts at -1. push: ++top THEN store; full -> print "Overflow ". pop: return THEN top--; empty -> print "Underflow " and return -1. peek: st[top] or -1.

void push(int val) {
    if(top==MAX-1){
        printf("Overflow ");
        return;
    }
    st[++top]=val;
}

int pop(void) {
    if(top==-1){
        printf("Underflow ");
        return -1;
    }
    return st[top--];
}

int peek(void) {
    return top==-1?-1:st[top];
}

Array queue

Circular buffer: indices wrap with % MAX; count tells full (MAX) from empty (0). front moves on dequeue, rear moves on enqueue.
Given types
#define MAX 5
int q[MAX];
int front=0,rear=-1,count=0;
enqueue + dequeue (circular array)

Circular: rear = (rear+1)%MAX, front = (front+1)%MAX, track count. Full -> print "Full ". Empty -> print "Empty " and return -1.

void enqueue(int val) {
    if(count==MAX){
        printf("Full ");
        return;
    }
    rear=(rear+1)%MAX;
    q[rear]=val;
    count++;
}

int dequeue(void) {
    if(count==0){
        printf("Empty ");
        return -1;
    }
    int v=q[front];
    front=(front+1)%MAX;
    count--;
    return v;
}

Hash: chaining

Index = key % SIZE. Collision = same bucket list. Delete: compare with == (not =), relink prev/head, free, then STOP using the node. Average O(1), worst O(n).
Given types
#define SIZE 10
typedef struct Node{int key,val;struct Node*next;}Node;
Node*hastable[SIZE]={NULL};
int has(int key){return key%SIZE;}
insertchaining

Insert at the head of the bucket list hastable[has(key)].

void insertchaining(int key,int val) {
    int i=has(key);
    Node*n=malloc(sizeof(Node));
    n->key=key;
    n->val=val;
    n->next=hastable[i];
    hastable[i]=n;
}
deleteChaining

Delete the node with this key from its bucket (head or middle), free it, then STOP. Not found: do nothing.

void deleteChaining(int key) {
    int i=has(key);
    Node*c=hastable[i],*prev=NULL;
    while(c&&c->key!=key){
        prev=c;
        c=c->next;
    }
    if(!c)return;
    if(prev)prev->next=c->next;
    else hastable[i]=c->next;
    free(c);
}
searchChaining

Walk only the bucket hastable[has(key)]; return the node or NULL.

Node *searchChaining(int key) {
    Node*c=hastable[has(key)];
    while(c&&c->key!=key)c=c->next;
    return c;
}
freetable

Free every node of every bucket and set each bucket back to NULL.

void freetable(void) {
    for(int i=0;i<SIZE;i++){
        Node*c=hastable[i];
        while(c){
            Node*t=c;
            c=c->next;
            free(t);
        }
        hastable[i]=NULL;
    }
}
displaychaining

For every bucket print "[i]" then " key -> val" for each node in it, then a newline.

void displaychaining(void) {
    for(int i=0;i<SIZE;i++){
        printf("[%d]",i);
        for(Node*c=hastable[i];c;c=c->next)printf(" %d -> %d",c->key,c->val);
        puts("");
    }
}

Hash: probing

Probe (key+i)%SIZE. isocupied: 0 empty, 1 used, -1 deleted (tombstone). Search stops ONLY at 0 and skips -1. Delete marks -1, NEVER 0. Insert: update if key exists, else reuse the first tombstone seen.
Given types
#define SIZE 10
typedef struct Hashentry{int val,key,isocupied;}Hashentry;   // isocupied: 0 empty, 1 used, -1 deleted
Hashentry table[SIZE];
int hash(int key,int i){return (key+i)%SIZE;}
insertprobing

Linear probing: existing key -> update value; else use the first deleted slot seen (or the empty one found); full -> print "Table is full".

void insertprobing(int key,int value) {
    int fd=-1,x=0;
    for(int i=0;i<SIZE;i++){
        x=hash(key,i);
        if(table[x].isocupied==1&&table[x].key==key){
            table[x].val=value;
            return;
        }
        if(table[x].isocupied==-1&&fd==-1)fd=x;
        if(table[x].isocupied==0){
            if(fd!=-1)x=fd;
            table[x].key=key;
            table[x].val=value;
            table[x].isocupied=1;
            return;
        }
    }
    if(fd==-1){
        printf("Table is full");
        return;
    }
    table[fd].key=key;
    table[fd].val=value;
    table[fd].isocupied=1;
}
deleteprobing

Find the key (stop at an EMPTY slot), mark it DELETED (-1, never 0!). Return 1 if deleted, 0 if not found.

int deleteprobing(int key) {
    for(int i=0;i<SIZE;i++){
        int x=hash(key,i);
        if(table[x].isocupied==0)return 0;
        if(table[x].isocupied==1&&table[x].key==key){
            table[x].isocupied=-1;
            return 1;
        }
    }
    return 0;
}
searchprobing

Probe: stop at an EMPTY slot (0) -> NULL, SKIP deleted (-1), return &table[x] if used and key matches. Max SIZE steps.

Hashentry *searchprobing(int key) {
    for(int i=0;i<SIZE;i++){
        int x=hash(key,i);
        if(table[x].isocupied==0)return NULL;
        if(table[x].isocupied==1&&table[x].key==key)return &table[x];
    }
    return NULL;
}
displayprob

For each slot print "[i] key -> val", "[i] deleted" or "[i] empty" (one per line).

void displayprob(void) {
    for(int i=0;i<SIZE;i++){
        printf("[%d] ",i);
        if(table[i].isocupied==1)printf("%d -> %d",table[i].key,table[i].val);
        else if(table[i].isocupied==-1)printf("deleted");
        else printf("empty");
        puts("");
    }
}

Sorting

Bubble: inner loop to n-i-1, flag = early exit. Selection: swap the minimum into i. Insertion: start at i=1, shift right. Merge: temp arrays, <= keeps it stable. Complexity: bubble/selection/insertion O(n^2) (bubble+insertion O(n) best case); merge O(n log n) always, O(n) extra memory.
Given types
bublesort

Bubble sort with the early exit flag. Inner loop stops at n-i-1.

void bublesort(int arr[],int n) {
    for(int i=0;i<n-1;i++){
        int s=0;
        for(int j=0;j<n-i-1;j++){
            if(arr[j]>arr[j+1]){
                int t=arr[j];
                arr[j]=arr[j+1];
                arr[j+1]=t;
                s=1;
            }
        }
        if(!s)return;
    }
}
selectionsort

Find the minimum of the unsorted part, swap it to position i.

void selectionsort(int arr[],int n) {
    for(int i=0;i<n-1;i++){
        int m=i;
        for(int j=i+1;j<n;j++){
            if(arr[j]<arr[m])m=j;
        }
        int t=arr[m];
        arr[m]=arr[i];
        arr[i]=t;
    }
}
insertsort

Take arr[i], shift bigger elements right, drop key in the gap.

void insertsort(int arr[],int n) {
    for(int i=1;i<n;i++){
        int key=arr[i],j=i-1;
        while(j>=0&&arr[j]>key){
            arr[j+1]=arr[j];
            j--;
        }
        arr[j+1]=key;
    }
}
merge + mergeSort

Write BOTH: merge copies the halves to temp arrays (malloc+free) and merges with <= (stable); mergeSort splits at mid and recurses.

void merge(int arr[],int left,int mid,int right) {
    int n1=mid-left+1,n2=right-mid;
    int*L=malloc(n1*sizeof(int)),*R=malloc(n2*sizeof(int));
    for(int i=0;i<n1;i++)L[i]=arr[left+i];
    for(int j=0;j<n2;j++)R[j]=arr[mid+1+j];
    int i=0,j=0,k=left;
    while(i<n1&&j<n2){
        if(L[i]<=R[j])arr[k++]=L[i++];
        else arr[k++]=R[j++];
    }
    while(i<n1)arr[k++]=L[i++];
    while(j<n2)arr[k++]=R[j++];
    free(L);
    free(R);
}

void mergeSort(int arr[],int left,int right) {
    if(left<right){
        int mid=(left+right)/2;
        mergeSort(arr,left,mid);
        mergeSort(arr,mid+1,right);
        merge(arr,left,mid,right);
    }
}
